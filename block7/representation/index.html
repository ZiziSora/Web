const rows = await tx.$queryRaw(Prisma.sql`
  SELECT id, name, ${extra}
  FROM ${this.table(resource)}
  WHERE id > ${query.after}
    AND "searchName" LIKE ('%' || ${term} || '%')
  ORDER BY id ASC LIMIT ${query.limit + 1}
`);
const hasMore = rows.length > query.limit;
const items = rows.slice(0, query.limit);
return {
  items, hasMore,
  nextCursor: hasMore ? items.at(-1)!.id : null
};

RelationField.tsx · khung nguyên liệuLược trích · xuống dòng lại
if (!infinite || !s.hasMore || s.loading ||
    s.error || !sentinel.current ||
    search.trim() !== q) return;
const observer = new IntersectionObserver(
  entries => {
    if (entries.some(entry => entry.isIntersecting))
      load.current();
  },
  { root: root.current, threshold: 0.1 }
);
observer.observe(sentinel.current);
return () => observer.disconnect();

Network cho thấy server chỉ trả một phần danh sách

Mở web localhost:5176 → F12 → Network. Theo dõi Fetch/XHR để đọc JSON, rồi đối chiếu cột Type hoặc bộ lọc Doc.

  1. Clear log, mở thực đơn, chuyển trang 2.
    Chọn request dishes có query; xem Payload → Query String Parameters.
  2. Chỉ ra page=2 và pageSize=6.
    Response có tối đa 6 món, total và số trang. Không có mảng 1.000 món để client tự cắt.
  3. Form món → Tải thêm / cuộn trong khung.
    Request options có limit=6; lần sau có after lấy từ nextCursor. Response lô cuối có hasMore=false.
  4. Đổi filter hoặc sort.
    Request mới mang đủ điều kiện, về trang 1; thao tác sau khi app đã tải không tạo document navigation.
